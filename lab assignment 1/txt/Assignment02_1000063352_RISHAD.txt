import java.util.Scanner;

public class q2{
    public static void main(String[] stopped) {
        Scanner scanner = new Scanner(System.in);
        System.out.print("Age: ");
        int age = scanner.nextInt();

        System.out.print("Electricity Consumed: ");
        double electricityConsumed = scanner.nextDouble();
        double baseRate = 15.0;
        double baseBill = electricityConsumed * baseRate;

        double discountPercent = 0.0;
        if (age < 18) {
            discountPercent = 20.0;
        } else if (age > 60) {
            discountPercent = 10.0; 
        } else {
            discountPercent = 0.0;
        }
        double billAfterDiscount = baseBill * (1 - (discountPercent / 100.0));

        if (electricityConsumed > 100) {
            billAfterDiscount = billAfterDiscount * 1.05;
        }

        System.out.printf("Final Bill: %.1f Taka\n", billAfterDiscount);

        scanner.close();
    }
}

import java.util.Scanner;

public class qn4 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int c = sc.nextInt();

        if (a == b && b == c) {
            System.out.println("This is a Equilateral triangle");
        } else if (a == b || b == c || a == c) {
            System.out.println("This is a Isosceles triangle");
        } else {
            System.out.println("This is a Scalene triangle");
        }
    }
}

import java.util.Scanner;

public class Question5 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("Enter the amount the customer need to pay(Taka)");
        int bill = sc.nextInt();
        System.out.println("Enter the amount, customer gave(Taka)");
        int given = sc.nextInt();

        if (given < bill) {
            System.out.println("Please pay " + (bill - given) + " taka more.");
        } else {
            int change = given - bill;
            System.out.println("The returned amount is " + change + " taka.");
            if (change > 0) {
                System.out.println("100 taka note: " + change / 100);
                change %= 100;
                System.out.println("50 taka note: " + change / 50);
                change %= 50;
                System.out.println("20 taka note: " + change / 20);
                change %= 20;
                System.out.println("10 taka note: " + change / 10);
                change %= 10;
                System.out.println("5 taka coin: " + change / 5);
                change %= 5;
                System.out.println("2 taka coin: " + change / 2);
                change %= 2;
                System.out.println("1 taka coin: " + change);
            }
        }
    }
}
